فریم‌ورک نسل جدید ربات‌های پیام‌رسان بله

توسعه پرسرعت، همزمان و ایمن با GoBale

فریم‌ورک GoBale یک بسته ماژولار، با کارایی فوق‌العاده بالا و الگوی طراحی زنجیره‌ای (Fluent Builder) برای زبان Go است که تمامی قابلیت‌های پیام‌رسان بله را با نهایت امنیت، دیتابیس بدون افت دیتای WAL، سپر ضد کرش یونیکد و ابزارهای مانیتورینگ زنده در اختیارتان می‌گذارد.

شروع کار در ۵ دقیقه مشاهده کاتالوگ متدها
عملکرد فوق سریع O(1)
بهره‌گیری از کلاینت HTTP بهینه با Connection Pooling، فشرده‌سازی خودکار Gzip و دیتابیس حافظه‌محور WAL.
سپر امنیتی چندبعدی
تشخیص و خنثی‌سازی خودکار بمب‌های یونیکد، کاراکترهای Zalgo، متون مخرب و جلوگیری قطعی از کرش بات.
معماری زنجیره‌ای (Fluent)
سینتکس تمیز و خوانا برای ارسال پیام، مدیا، کیبوردها، تنظیمات و میدلورها با متد Go().
دیتابیس درونی WAL
سیستم ماندگاری وضعیت کاربران و تنظیمات با لاگ ترتیبی باینری، بازیابی خودکار و Compactor دوره‌ای.
Chapter 01

فصل ۱: مقدمه، ویژگی‌ها و شروع سریع

آشنایی با ساختار پکیج GoBale، نحوه نصب، تنظیم متغیرهای محیطی و راه‌اندازی اولین ربات انعکاس‌دهنده (Echo Bot).

۱.۱ پیش‌نیازها و نصب پکیج

برای شروع به زبان برنامه‌نویسی Go نسخه 1.20 یا بالاتر نیاز دارید. بسته را با اجرای دستور زیر در ترمینال اضافه کنید:

ترمینال - نصب پکیج
bash 
go get github.com/PHX-Go/GoBale

یا برای دریافت آخرین تغییرات برنچ اصلی مخزن:

ترمینال - نصب نسخه اصلی
bash 
go get github.com/PHX-Go/GoBale@main

۱.۲ پیکربندی متغیرهای محیطی (.env)

یک فایل با نام .env در پوشه ریشه پروژه خود ساخته و توکن بات و شناسه ادمین را وارد کنید:

فایل .env
env 
BALE_TOKEN=123456789:ABCdefGHIjklMNOpqrsTUVwxyz\nADMIN_ID=987654321

۱.۳ نمونه کد کامل اکو بات (Echo Bot)

کد زیر یک نمونه اجرایی کامل با پشتیبانی از فشرده‌سازی Gzip و پردازش دستورات است:

main.go
go 
package main

import (
	"log"

	gobale "github.com/PHX-Go/GoBale"
)

func main() {
	// بارگذاری متغیرهای محیطی از فایل .env
	gobale.Env().Go()

	token := gobale.GetEnv[string]("BALE_TOKEN")
	adminID := gobale.GetEnv[int64]("ADMIN_ID")

	// مقداردهی اولیه بات با تنظیم شناسه ادمین و فشرده‌سازی Gzip
	bot, err := gobale.New(token).
		Admin(adminID).
		Gzip().
		Go()

	if err != nil {
		log.Fatalf("خطا در راه‌اندازی ربات: %v", err)
	}

	// تعریف هندلر دستور /start
	bot.On().Cmd("start").Do(func(c *gobale.Ctx) {
		c.Send().
			Text("به ربات GoBale خوش آمدید! هر پیامی ارسال کنید تکرار می‌شود.").
			Go()
	})

	// تعریف هندلر دریافت کلیه پیام‌های متنی
	bot.On().Msg().Do(func(c *gobale.Ctx) {
		c.Send().
			Text("پیام شما: " + c.Text()).
			Go()
	})

	// اجرای ربات به روش Polling
	log.Println("ربات در حال اجرا است...")
	bot.Run().Polling().Go()
}
بهینه‌سازی مصرف شبکه

متد .Gzip() مصرف پهنای باند سرور را در ارسال و دریافت دیتا تا بیش از ۷۰ درصد کاهش می‌دهد.

۱.۴ دیاگرام چرخه پردازش رویدادها

چرخه پردازش Update در GoBale
سرور بله (Bale API)
←
Worker Pool / Receiver
←
سپر ضد کرش یونیکد
←
پایپ‌لاین میدلورها (Use)
←
موتور روتر (On)
←
هندلر کاربر (Do)

فصل ۲: معماری هسته و روش‌های اجرا

بررسی گزینه‌های سازنده gobale.New() و ۳ روش اجرای Polling، Webhook و Native WebSocket.

۲.۱ سازنده ربات و گزینه‌های زنجیره‌ای

متد پیکربندی نوع ورودی توضیح عملکرد
.Admin(ids ...int64)int64ثبت شناسه‌های عددی ادمین‌های بات برای اعتبارسنجی‌ها.
.Gzip()-فعال‌سازی فشرده‌سازی ترنزیت دیتا.
.Timeout(d time.Duration)time.Durationتایم‌اوت درخواست‌های شبکه به سرور بله (پیش‌فرض: 30s).
.Workers(n int)intتعداد پردازنده‌های موازی در صف پردازش بات.
.Logger(l *slog.Logger)*slog.Loggerتزریق لاگر ساختاریافته اختصاصی.
.Database(db Storage)Storageتزریق دیتابیس اختصاصی یا WAL محلی.
.EventBus(eb *EventBus)*EventBusمتصل‌کردن سیستم انتشار و اشتراک رویدادها.

۲.۲ روش‌های سه‌گانه اجرا

روش‌های اجرا
go 
// ۱. اجرای Long Polling
bot.Run().Polling().
	Limit(100).
	Timeout(25).
	Go()

// ۲. اجرای Webhook
bot.Run().Webhook().
	Listen(":8443").
	URL("https://yourdomain.com/webhook").
	Secret("secret_key").
	Go()

// ۳. اجرای WebSocket Native
bot.Run().WebSocket().
	URL("wss://api.bale.ai/v1/stream").
	Reconnect(true).
	Go()

فصل ۳: موتور مسیریابی و هندلرها

نحوه تعریف مسیرها با bot.On()، استخراج آرگومان‌های دستور، فیلترهای Regex و رویدادهای دکمه شیشه‌ای.

۳.۱ فیلترهای رویداد

انواع فیلترها
go 
// دستورات با اسامی مستعار
bot.On().Cmd("help", "rahnama", "guide").Do(func(c *gobale.Ctx) {
	c.Send().Text("لیست راهنما...").Go()
})

// تطبیق متن با Regex
bot.On().Regex(`^کد:\s*(\d+)$`).Do(func(c *gobale.Ctx) {
	c.Send().Text("کد پیگیری دریافت شد.").Go()
})

// دریافت انواع رسانه
bot.On().Photo().Do(func(c *gobale.Ctx) { c.Reply().Text("عکس دریافت شد.").Go() })
bot.On().Video().Do(func(c *gobale.Ctx) { c.Reply().Text("ویدیو دریافت شد.").Go() })
bot.On().Voice().Do(func(c *gobale.Ctx) { c.Reply().Text("ویس دریافت شد.").Go() })
bot.On().Document().Do(func(c *gobale.Ctx) { c.Reply().Text("فایل دریافت شد.").Go() })

// کلیک دکمه‌های شیشه‌ای
bot.On().Callback("menu_settings").Do(func(c *gobale.Ctx) {
	c.Answer().Text("ورود به تنظیمات...").Go()
})

۳.۲ استخراج آرگومان‌های دستورات

استخراج آرگومان‌ها
go 
bot.On().Cmd("ban").AdminOnly().Do(func(c *gobale.Ctx) {
	// استخراج مستقیم با تعیین مقدار پیش‌فرض
	targetID := c.ArgInt(0, 0)
	reason := c.ArgString(1, "بدون دلیل مشخص")

	// اسکن همزمان با ScanArgs
	var userID int64
	var banReason string
	if err := c.ScanArgs(&userID, &banReason); err != nil {
		c.Reply().Text("فرمت: /ban 12345 دلیل").Go()
		return
	}
})

فصل ۴: موتور کانتکست (Context Engine - Ctx)

متدهای کمکی بررسی چت، هویت کاربر، پیام‌های خودتخریب‌گر و کنترل پایپ‌لاین.

۴.۱ متدهای استخراج هویت

متدخروجیتوضیحات
c.UserID()(int64, error)شناسه عددی کاربر فرستنده.

c.IsAdmin()boolبررسی دسترسی ادمین.
c.IsGroup()boolتشخیص گروه یا سوپرگروه.

اکشن‌ها و پیام موقت

فصل ۷: سپر امنیتی، میدلورها و AntiCrash

سپر محافظت در برابر متون مخرب Zalgo، بمب‌های یونیکد، میدلورهای ضد اسپم، ضد لینک و قفل رسانه.

۷.۱ سپر ضد کرش یونیکد (Zalgo Defense)

خنثی‌سازی خودکار تهدیدات متنی یونیکد و جلوگیری از کرش ربات در مواجهه با ورودی‌های مخرب:

فعال‌سازی AntiCrash
go 
// ثبت میدلور عمومی ضد کرش
bot.Use(gobale.AntiCrashMiddleware())

۷.۲ میدلورهای امنیتی گروه

میدلورهای محافظتی
go 
// ۱. ضد لینک با لیست سفید دامنه‌ها
bot.Use(gobale.AntiLink(nil, 5*time.Second, "ارسال لینک ممنوع است!", "ble.ir", "bale.ai"))

// ۲. ضد اسپم (حداکثر ۵ پیام در بازه ۳ ثانیه‌ای)
bot.Use(gobale.AntiSpam(nil, 5, 3*time.Second, "لطفاً اسپم نکنید!"))

// ۳. ضد کلمات نامناسب
bot.Use(gobale.AntiProfanity(nil, 5*time.Second, []string{"کلمه۱", "کلمه۲"}))

// ۴. ضد پیام‌های فوروارد شده
bot.Use(gobale.AntiForward(nil, 5*time.Second))

فصل ۸: موتور اخطار و مجازات پلکانی

سیستم ثبت و انقضای اخطارها، اعمال مجازات‌های خودکار (سکوت، اخراج، بن) و دستورات ادمین.

۸.۱ پیکربندی WarnEngine

موتور اخطار پلکانی
go 
engine := gobale.NewWarnEngine(bot).
	AddStep(gobale.Warn("اخطار ۱: قوانین را رعایت کنید.")).
	AddStep(gobale.Mute(2*time.Hour, "شما ۲ ساعت سایلنت شدید.")).
	AddStep(gobale.Kick("شما از گروه اخراج شدید.")).
	AddStep(gobale.Ban("شما به صورت دائمی مسدود شدید.")).
	Cooldown(24 * time.Hour) // کاهش خودکار اخطار پس از ۲۴ ساعت

// ثبت خودکار دستورات /warn, /unwarn, /warns
engine.AutoCommands()

// اتصال به میدلور ضد لینک
bot.Use(gobale.AntiLink(engine, 5*time.Second, "ارسال لینک ممنوع است!"))

فصل ۹: جریان‌های مکالمه‌ای و ویزاردها

طراحی فرم‌های چندمرحله‌ای تعاملی با اعتبارسنجی ورودی هر مرحله.

۹.۱ ساخت ویزارد مکالمه (WizardChain)

ویزارد ثبت‌نام
go 
bot.On().Cmd("register").Do(func(c *gobale.Ctx) {
	c.NewWizard().
		Step("نام و نام خانوادگی خود را وارد کنید:", func(ctx *gobale.Ctx) {
			ctx.SetData("name", ctx.Text())
		}).
		StepWithValidation(
			"شماره تماس ۱۱ رقمی خود را وارد کنید:",
			func(text string) bool { return len(text) == 11 },
			"شماره نامعتبر است! باید ۱۱ رقمی باشد.",
			func(ctx *gobale.Ctx) {
				ctx.SetData("phone", ctx.Text())
			},
		).
		OnComplete(func(ctx *gobale.Ctx) {
			ctx.Send().Text("ثبت‌نام با موفقیت تکمیل شد!").Go()
		}).
		Start()
})
